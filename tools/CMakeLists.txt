option(MARATHON_RECOMP_OPTIMIZE_TOOLS "Apply compiler optimizations to build tools." ON)
if (MARATHON_RECOMP_OPTIMIZE_TOOLS)
    if (WIN32)
        add_compile_options(/O2)
    else()
        add_compile_options(-O3)
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()

if (CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT MARATHON_RECOMP_HOST_TOOLS_ONLY)
    # When cross-compiling for iOS, host CLI tools (file_to_c, XenonRecomp, u8extract, XenosRecomp)
    # are built natively for macOS in advance via MARATHON_RECOMP_HOST_TOOLS_ONLY=ON.
    # Here we only include the cross-compiled target libraries and shader compiler definitions.
    set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)

    set(THIRDPARTY_ROOT "${MARATHON_RECOMP_TOOLS_ROOT}/XenonRecomp/thirdparty")
    set(XENONUTILS_ROOT "${MARATHON_RECOMP_TOOLS_ROOT}/XenonRecomp/XenonUtils")
    add_subdirectory("${THIRDPARTY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/XenonRecomp-thirdparty")
    add_subdirectory("${XENONUTILS_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/XenonUtils")

    set(XENOS_RECOMP_THIRDPARTY_ROOT "${MARATHON_RECOMP_TOOLS_ROOT}/XenosRecomp/thirdparty")
    add_subdirectory("${XENOS_RECOMP_THIRDPARTY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/XenosRecomp-thirdparty")
else()
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/bc_diff)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/file_to_c)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/fshasher)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/u8extract)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/x_decompress)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/XenonRecomp)
    add_subdirectory(${MARATHON_RECOMP_TOOLS_ROOT}/XenosRecomp)
    if (TARGET XenosRecomp)
        target_compile_definitions(XenosRecomp PRIVATE MARATHON_RECOMP)
    endif()
endif()
